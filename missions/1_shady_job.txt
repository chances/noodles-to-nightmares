{$CLEO .cm}

//==============================================
// Mission 1: Deliver The Goods; Easy Money
// Move some contraband from point A to point B
//
// Teach the player how to steal and drive cars
//==============================================
const REWARD = 800 // Fee for the job

$onMission = true
$isDying = false
$isWanted = false

Text.PrintBigString("Easy Money", {time} 5000, TextStyle.BottomRight)
wait 7500

// QUESTION: Who is speaking?
Text.PrintString("I hear you're meek like a mouse. I need someone discreet.", 3500)
wait 3750
Text.PrintString("I'll call you later at your payphone.", 4000)
wait 4250

// Wait for the player to get near the payphone
while true
  wait 0

  if
    Player.LocateAnyMeans2D($player, 1168.33, -414.141, 20, 20, false)
  then
    // Ring the payphone
    Sound.AddContinuous(1168.33, -414.141, 22.8, ScriptSound.SoundPayphoneRinging, $payphoneSfx)
    break
  end
end

// TODO: From here on forward, NO wanted level over one star!

// Wait for the player to "answer" the phone
while true
  wait 0

  if and
    Player.LocateAnyMeans2D($player, 1168.33, -414.841, 2, 2, true)
    Player.IsLiftingAPhone($player)
  then
    wait 250
    Sound.Remove($payphoneSfx)
    // TODO: Use a cinematic camera?
    break
  end
end

Text.PrintString("Pick up the package at the docks, deliver it to the projects on the north side.", 4500)
wait 4750
Text.PrintString("Keep a low profile. We don't want the cops sniffin' around.", 3500)
wait 3750
Text.PrintString("You don't have much time. Don't fuck this up.", 3000)
wait 3250

// TODO: Start a two minute countdown

// Spawn the drugs pickup
$package = Pickup.Create(1321, PickupType.OnStreet, 1316.96, -1088.5, 11.1705)
$pickupBlip = Blip.AddForPickup($package)
// Wait for the package to be picked up
while true
  wait 0

  if Pickup.HasBeenCollected($package)
  then
    Blip.Remove($pickupBlip)
    Pickup.Remove($package)
    break
  end
end

// Wait for the player to "deliver" the package to the projects
$deliveryBlip = Blip.AddForCoord(966.364, -133.515, 3.8)
while true
  wait 0

  if
    Player.LocateAnyMeans3D($player, 966.364, -133.515, 3.8, 2.5, 2.5, 2.5, true)
  then
    Blip.Remove($deliveryBlip)
    break
  end
end

// TODO: Refactor above from no-wanted-level to here into a main mission loop

// The player successfully delivered the package, pass the mission
Audio.PlayMissionPassedTune(1)
register_mission_passed 'NooM_1'
Player.AddScore($player, REWARD)
Text.PrintWithNumberBig('M_PASS', REWARD, 5000, TextStyle.Middle)
wait 5500

// TODO: Send this text on the pager
Text.PrintString("Here's your cut. There may be more work soon. I'll be in touch.", 4000)
wait 4250

Mission.Finish()
$onMission = false
$missionIndex = $missionIndex + 1

05DC: terminate_custom_thread
