/// Gracefully remove Sal if he's far away or 7.5 seconds have passed since the cutscene ended
function tryRemoveSal()
  if Char.LocateAnyMeansChar2D($sal, $playerChar, 50, 50, false)
  then
    $isSalNearby = true
  else
    $isSalNearby = false
  end
  if or
    $isSalNearby == false
    TIMERA > 7500
  then
    Char.RemoveElegantly($sal)
  end
end
