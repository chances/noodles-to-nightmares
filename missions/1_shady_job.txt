/// Gracefully remove Sal if he's far away or 7.5 seconds have passed since the cutscene ended
function tryRemoveSal()
  Char.MarkAsNoLongerNeeded($sal)

  if Char.LocateAnyMeansChar2D($sal, $playerChar, 50.0, 50.0, false)
  then
    $isSalNearby = true
  else
    $isSalNearby = false
  end
  if
    $isSalNearby == false
  then
    Char.RemoveElegantly($sal)
  end
  if
    TIMERA > 7500
  then
    Char.Delete($sal)
    $sal = 0
  end
end
