// Wait until player character is valid
while true
  wait 250 ms
  if
    $player > 0
  then

  end

  wait 66 ms // 15Hz

  if
    $player > 0
  then
    $playerChar = Player.GetChar($player)
    Char.GetCoordinates($playerChar, 0@, 1@, 2@)
    Char.GetHeading($playerChar, 3@)

    4@ = 0
    if
      0AB0: is_key_pressed 0x77 // F8
    then
      4@ = 1
    end

    0B30: 8089 0@ 1@ 2@ 3@ 4@
  end
end
