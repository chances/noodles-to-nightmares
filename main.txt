/// Trigger the next mission, if any
function tryTriggerMission()
  if and
    $onMission == false
    $missionIndex < MAX_MISSION_INDEX
    Player.CanStartMission($player)
  then
    // Try to trigger the next mission
    switch $missionIndex
      case 0
        clearUI()
        register_mission_given
        0600: START_CUSTOM_THREAD "0_tutorial.cm"
      case 1
        // Transition: Add a contect blip to Punk Noodles
        if $NoodlesBlip == 0
        then
          // Add a contect blip to Punk Noodles
          Blip.AddSpriteForContactPoint(1031.13, -654.956, 14.0094, RadarSprite.Asuka, $NoodlesBlip)
        end

        // Trigger: Start the mission inside Punk Noodles
        if Player.LocateStoppedOnFoot3D($player, 1031.13, -654.956, 14.0094, 2.5, 2.5, 2.5, true)
        then
          clearUI()
          register_mission_given
          0600: START_CUSTOM_THREAD "1_shady_job.cm"
        end
      default
        clearUI()
        Text.PrintHelpString("You're done with the game already?")
        wait 2500
    end
    // Player is NOT on a mission; do some chores:
    else
      // Ensure the player's safehouse blip is shown
      if $SafeBlip == 0
      then
        $SafeBlip = Blip.AddSpriteForCoord(1165.0, -398.5, 25.8, RadarSprite.Savehouse)
      end
  end
end

function monitorShortcutKey_P()
  if or
    Pad.IsKeyPressed(KeyCode.P)
    // FIXME: This might conflict with the change camera shortcut
    Pad.IsButtonPressed(PadId.Pad1, Button.Select)
  then
    Game.ActivateSaveMenu()

    // The user saved their game, so wait for the save procedure to finish
    while true
      wait 0
      if
        Game.HasSaveGameFinished()
      then
        break
      end
    end
  end
end

function monitorShortcutKey_T()
  if Pad.IsKeyPressed(KeyCode.T)
  then
    Text.PrintHelpString("Beaming you to the front door of Rosa's apartment")
    wait 3250

    // Teleport player to front door of Rosa's apartment
    teleport(1170.372, -416.70, 23.178)
  end
end
