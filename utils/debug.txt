//================================
//------- Functions: Debug -------
//================================
// Tools that stream readings to the mission editor over UDP (plugins/sockets, opcode 0B31), so values are read
// off the editor's debugger pane instead of the screen or a file.
// Needs sockets.cleo in CLEO\CLEO_PLUGINS and constants.txt for TELEMETRY_PORT and the DEBUG_* channels.
//
// Notes:
// - A caller passes a DEBUG_* constant from constants.txt as the channel. Adding a log point is one const there.
// - The plugin sends a reading only when it changes (or once a second), so calling every frame costs a compare.
// - Nothing is written to disk and nothing waits for a listener: with no editor open the packets are dropped.
// - The editor's debugger pane and `deno task run` read the DEBUG_* names out of constants.txt (scripts/telemetry.ts).

// Opcode 0B31 is declared in ./functions.txt as udp_send_debug, which every script includes first:
//   0B31: port, channel, kind, value

// Reading kinds understood by opcode 0B31
const DEBUG_KIND_INT = 0
const DEBUG_KIND_FLOAT = 1
const DEBUG_KIND_COUNT = 2
const DEBUG_KIND_TRACE = 3

/// Logs a float reading on the given DEBUG_* channel
function debugLogFloat(point: int, value: float)
  0B31: TELEMETRY_PORT point DEBUG_KIND_FLOAT value
end

/// Logs an int reading on the given DEBUG_* channel
function debugLogInt(point: int, value: int)
  0B31: TELEMETRY_PORT point DEBUG_KIND_INT value
end

/// Adds one to the counter on the given DEBUG_* channel; the plugin keeps the count
function debugCount(point: int)
  0B31: TELEMETRY_PORT point DEBUG_KIND_COUNT 0
end

/// Records a step of a trace on the given DEBUG_* channel; every call is sent, so the editor sees the order
function debugTrace(point: int, step: int)
  0B31: TELEMETRY_PORT point DEBUG_KIND_TRACE step
end

/// Logs a character's heading, in degrees, on the given DEBUG_* channel
function debugLogHeading(point: int, ped: Char)
  float heading = Char.GetHeading(ped)
  debugLogFloat(point, heading)
end
