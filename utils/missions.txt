/// Inverts the given heading, e.g. 90° becomes 270°.
function invertHeading(heading: float): float
  // Note: Sanny compiles a three-operand float sum as INT_ADD (0A8E), so add in place
  float opposite = heading
  opposite += 180.0
  if opposite >= 360.0
  then
    opposite -= 360.0
  end
  // Note: `return` is not supported in GTA 3!
  // See https://docs.sannybuilder.com/language/functions#return-from-function
  // Note: The first `cleo_return` argument is the number of returned values, e.g. `1`
  cleo_return 1 opposite
end

/// Ends a mission as passed: advances the story and terminates this mission thread.
function finishMission()
  Mission.Finish()
  $onMission = false
  $missionIndex = $missionIndex + 1

  // NOTE: 05DC for GTA III CLEO, in lieu of `terminate_this_custom_script` (San Andreas)
  05DC: terminate_custom_thread
end

//================================
//-------- Functions: GUI --------
//================================

function clearUI()
  Text.ClearSmallPrints()
  Text.ClearPrints()
end

/// Ensure the player's safehouse blip is shown, if they've completed the tutorial.
function ensureSafehouseBlip()
  if and
    $missionIndex > 0
    $SafeBlip == 0 // Only create the blip if it doesn't already exist
  then
    $SafeBlip = Blip.AddSpriteForCoord(SAFEHOUSE_X, SAFEHOUSE_Y, SAFEHOUSE_Z, RadarSprite.Savehouse)
  end
end

/// Ensure Lee's van blip is shown.
function ensureVanBlip()
  if and
    $VanBlip == 0 // Only create the blip if it doesn't already exist
    $missionIndex > 0
    $myCar > 0
    $myCarModel == #RUMPO // Only show the blip for the van model
  then
    $VanBlip = Blip.AddForCar($myCar)
    Car.MakeABitStronger($myCar, true)
  end
end

function saveGame() 
  Game.ActivateSaveMenu()
  while true
    wait 0
    if Game.HasSaveGameFinished()
    then
      break
    end
  end
end
